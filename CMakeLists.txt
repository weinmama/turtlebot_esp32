cmake_minimum_required(VERSION 3.19)
project(turtlebot_picoros_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PICOROS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PICOROS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(Z_FEATURE_UNSTABLE_API 1 CACHE STRING "" FORCE)
set(Z_FEATURE_LINK_SERIAL 1 CACHE STRING "" FORCE)
set(Z_FEATURE_AUTO_RECONNECT 0 CACHE STRING "" FORCE)
set(FRAG_MAX_SIZE 4096 CACHE STRING "" FORCE)
set(BATCH_UNICAST_SIZE 2048 CACHE STRING "" FORCE)
set(USER_TYPE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/include/robot_types.h" CACHE STRING "" FORCE)
add_subdirectory(thirdparty/picoros EXCLUDE_FROM_ALL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(patched_picoros "${CMAKE_CURRENT_BINARY_DIR}/picoros-patched/picoros.c")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_picoros.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/picoros/src/picoros.c"
  "${patched_picoros}" COMMAND_ERROR_IS_FATAL ANY)
set_property(TARGET picoros PROPERTY SOURCES "${patched_picoros}")
target_include_directories(picoros PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
set(patched_rx "${CMAKE_CURRENT_BINARY_DIR}/zenoh-patched/rx.c")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_zenoh.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/picoros/thirdparty/zenoh-pico/src/transport/unicast/rx.c"
  "${patched_rx}" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(zenoh_sources zenohpico_static SOURCES)
list(FILTER zenoh_sources EXCLUDE REGEX "transport/unicast/rx\\.c$")
set_property(TARGET zenohpico_static PROPERTY SOURCES "${zenoh_sources};${patched_rx}")
# Upstream leaves locals/parameters unused when AUTO_RECONNECT is disabled.
target_compile_options(zenohpico_static PRIVATE -Wno-error=unused-variable -Wno-error=unused-parameter)
add_library(robot_transport src/robot_transport.cpp)
option(ROBOT_TRANSPORT_DIAGNOSTICS "Enable verbose timing diagnostics" OFF)
if(ROBOT_TRANSPORT_DIAGNOSTICS)
  target_compile_definitions(robot_transport PUBLIC ROBOT_TRANSPORT_DIAGNOSTICS=1)
endif()
target_include_directories(robot_transport PUBLIC src include)
target_link_libraries(robot_transport PUBLIC picoros picoserdes pthread)
target_compile_options(robot_transport PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
enable_testing()
add_executable(test_picoros_uart test/test_picoros_uart.cpp)
target_include_directories(test_picoros_uart PRIVATE test/uart_stubs include)
target_link_libraries(test_picoros_uart PRIVATE pthread)
target_compile_options(test_picoros_uart PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME picoros_uart COMMAND test_picoros_uart)
set_tests_properties(picoros_uart PROPERTIES TIMEOUT 5)
add_executable(test_picoros_clock_ns test/test_picoros_clock_ns.c)
target_include_directories(test_picoros_clock_ns PRIVATE include)
target_compile_options(test_picoros_clock_ns PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME picoros_clock_ns COMMAND test_picoros_clock_ns)
foreach(name gyro_calibration picoros_time_sync picoros_command_guard encoder_feedback_guard encoder_pulse_filter encoder_speed_estimator)
  add_executable(test_${name} test/test_${name}.cpp)
  target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
target_include_directories(test_encoder_feedback_guard PRIVATE include)
target_include_directories(test_encoder_pulse_filter PRIVATE include)
target_include_directories(test_encoder_speed_estimator PRIVATE include)
add_executable(transport_fixture test/transport_fixture.cpp)
target_link_libraries(transport_fixture PRIVATE robot_transport)
add_executable(test_cdr test/test_cdr.cpp)
target_link_libraries(test_cdr PRIVATE picoserdes)
target_include_directories(test_cdr PRIVATE include)
target_compile_options(test_cdr PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME cdr COMMAND test_cdr)
add_executable(test_batched_frames test/test_batched_frames.c)
target_link_libraries(test_batched_frames PRIVATE zenohpico::lib)
target_compile_options(test_batched_frames PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME batched_frames COMMAND test_batched_frames)
